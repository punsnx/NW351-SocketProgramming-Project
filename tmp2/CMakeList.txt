cmake_minimum_required(VERSION 3.16)
project(rudp LANGUAGES CXX)

# --- Global settings
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(ENABLE_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)

# --- Sources
set(RUDP_CORE_SOURCES
    src/packet.cpp
    src/checksum.cpp
    src/timer.cpp
    src/window.cpp
    src/socket.cpp
    src/protocol.cpp
)

# --- Core library (pure logic)
add_library(rudp_core STATIC ${RUDP_CORE_SOURCES})
target_include_directories(rudp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Warnings
if (MSVC)
  target_compile_options(rudp_core PRIVATE /W4 /permissive-)
  if (ENABLE_WARNINGS_AS_ERRORS)
    target_compile_options(rudp_core PRIVATE /WX)
  endif()
else()
  target_compile_options(rudp_core PRIVATE -Wall -Wextra -Wpedantic)
  if (ENABLE_WARNINGS_AS_ERRORS)
    target_compile_options(rudp_core PRIVATE -Werror)
  endif()
endif()

# --- Single binary entrypoint
add_executable(rudp src/main.cpp)
target_link_libraries(rudp PRIVATE rudp_core)

# --- Install (optional)
include(GNUInstallDirs)
install(TARGETS rudp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

# --- (Optional) ctest hook
include(CTest)


mkdir build && cd build
cmake ..
cmake --build . -j
./rudp server 9000 65536
./rudp client 127.0.0.1 9000 big.bin 65536
